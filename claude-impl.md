## Data Models

### `department_model.dart`

- `fromJson`: `id`, `name` as `(json['x'] as String? ?? '').trim()`
- `toJson`: `{ 'id': id, 'name': name }`

### `book_model.dart`

- `fromJson`: all string fields safe-cast + trim
- `department`: `DepartmentModel.fromJson(json['department'] as Map? ?? {})`
- `articles`: `(json['articles'] as List? ?? []).map(ArticleModel.fromJson).toList()`
- `toJson`: all fields + `department: deptModel.toJson()` + `articles: [...toJson()]`

### `article_model.dart`

- `fromJson`:
  - `book_ids`: if `List` → cast; if `String` → split on `/*-*/`; else `[]`
  - all other fields safe-cast + trim
- `toJson`: `book_ids` joined with `/*-*/`, all others as-is

## Google Apps Script API

Base URL: `dotenv.env['APPS_SCRIPT_URL']`
All requests: `GET ?action=...&lang=ar|en&...`

| Action | Params | Returns |
|---|---|---|
| `getAllBooks` | `page`, `lang` | `{ books, total, page, has_more }` |
| `searchById` | `id`, `lang` | `{ book: { ...articles[] } }` or null |
| `search` | `query`, `lang` | `{ books: [...] }` each with articles |
| `getBooksByAuthor` | `author`, `lang` | `{ books: [...] }` |
| `getBooksByDepartment` | `department` (id e.g "D1"), `lang` | `{ books: [...] }` |
| `getDepartments` | `lang` | `{ departments: [{id, name}] }` |

`department` in every book response:

```json
{ "id": "D1", "name": "كهرباء" }
```

### `DepartmentCubit` — Factory HydratedCubit

```dart
enum DepartmentStatus { initial, loading, success, empty, error }

class DepartmentState extends Equatable {
  final DepartmentStatus status;
  final List<Department> departments;
  final List<Book> books;           // books for selected department
  final Department? selected;       // currently selected department
  final String errorMessage;
  // copyWith, props
  // fromJson: restore departments only (reset books + selected)
  // toJson: persist departments list only
}

class DepartmentCubit extends HydratedCubit<DepartmentState> {
  final GetDepartments getDepartmentsUseCase;
  final GetBooksByDepartment getBooksByDepartmentUseCase;

  Future<void> loadDepartments() async {
    emit(state.copyWith(status: DepartmentStatus.loading));
    try {
      final depts = await getDepartmentsUseCase();
      emit(state.copyWith(
        status: depts.isEmpty ? DepartmentStatus.empty : DepartmentStatus.success,
        departments: depts,
      ));
    } on Failure catch (f) {
      emit(state.copyWith(status: DepartmentStatus.error, errorMessage: f.message));
    }
  }

  Future<void> loadBooksByDepartment(Department dept) async {
    emit(state.copyWith(status: DepartmentStatus.loading, selected: dept, books: []));
    try {
      final books = await getBooksByDepartmentUseCase(dept.id);
      emit(state.copyWith(
        status: books.isEmpty ? DepartmentStatus.empty : DepartmentStatus.success,
        books: books,
      ));
    } on Failure catch (f) {
      emit(state.copyWith(status: DepartmentStatus.error, errorMessage: f.message));
    }
  }
}
```

### `SearchCubit` — Factory HydratedCubit

```dart
enum SearchStatus { initial, loading, success, empty, error }
enum SearchFilter { all, books, articles }

class SearchState extends Equatable {
  final SearchStatus status;
  final String query;
  final List<Book> results;
  final SearchFilter filter;
  final String errorMessage;
  // copyWith, props
  // fromJson: restore query + results + filter
  // toJson: persist same
}

class SearchCubit extends HydratedCubit<SearchState> {
  final SearchBooks searchBooksUseCase;
  Timer? _debounce;

  void search(String query) {
    _debounce?.cancel();
    if (query.trim().isEmpty) {
      emit(state.copyWith(status: SearchStatus.initial, query: '', results: []));
      return;
    }
    emit(state.copyWith(status: SearchStatus.loading, query: query));
    _debounce = Timer(const Duration(milliseconds: 500), () async {
      try {
        final results = await searchBooksUseCase(query);
        emit(state.copyWith(
          status: results.isEmpty ? SearchStatus.empty : SearchStatus.success,
          results: results,
        ));
      } on Failure catch (f) {
        emit(state.copyWith(status: SearchStatus.error, errorMessage: f.message));
      }
    });
  }

  void setFilter(SearchFilter filter) => emit(state.copyWith(filter: filter));
```

### `ScannerCubit` — Factory HydratedCubit (no persistence)

```dart
enum ScannerStatus { scanning, loading, found, notFound, error }

class ScannerState extends Equatable {
  final ScannerStatus status;
  final Book? book;
  final String errorMessage;
  // copyWith, props
  // fromJson: return null (always reset)
  // toJson: return null
}

class ScannerCubit extends HydratedCubit<ScannerState> {
  final GetBookById getBookByIdUseCase;
  bool _isProcessing = false;

  // Default test ID when barcode scan is unavailable
  static const String _testId = '11797998';

  Future<void> onBarcodeDetected(String? rawValue) async {
    if (_isProcessing) return;
    _isProcessing = true;

    // Use scanned value if valid, otherwise use test ID
    final id = (rawValue != null && rawValue.trim().isNotEmpty)
        ? rawValue.trim()
        : _testId;

    emit(state.copyWith(status: ScannerStatus.loading));
    try {
      final book = await getBookByIdUseCase(id);
      if (book == null) {
        emit(state.copyWith(status: ScannerStatus.notFound));
      } else {
        emit(state.copyWith(status: ScannerStatus.found, book: book));
      }
    } on Failure catch (f) {
      emit(state.copyWith(status: ScannerStatus.error, errorMessage: f.message));
    } finally {
      _isProcessing = false;
    }
  }

  void reset() {
    _isProcessing = false;
    emit(const ScannerState());
  }
}
```

### `FavoritesCubit` — Singleton HydratedCubit

```dart
enum FavoritesStatus { initial, success, empty }

class FavoritesState extends Equatable {
  final FavoritesStatus status;
  final List<Book> favorites;
  // copyWith, props
  // fromJson: restore favorites via BookModel.fromJson
  // toJson: persist as list of BookModel.toJson
}

class FavoritesCubit extends HydratedCubit<FavoritesState> {
  void addFavorite(Book book) {
    if (isFavorite(book.id)) return;
    final updated = [...state.favorites, book];
    emit(state.copyWith(
      favorites: updated,
      status: FavoritesStatus.success,
    ));
  }

  void removeFavorite(String id) {
    final updated = state.favorites.where((b) => b.id != id).toList();
    emit(state.copyWith(
      favorites: updated,
      status: updated.isEmpty ? FavoritesStatus.empty : FavoritesStatus.success,
    ));
  }

  bool isFavorite(String id) => state.favorites.any((b) => b.id == id);
}
```

## Dio Client — Lang Interceptor

```dart
class DioClient {
  late final Dio _dio;

  DioClient() {
    _dio = Dio(BaseOptions(
      baseUrl: dotenv.env['APPS_SCRIPT_URL'] ?? '',
      connectTimeout: const Duration(seconds: 15),
      receiveTimeout: const Duration(seconds: 15),
    ));
    _dio.interceptors.add(_LangInterceptor());
    _dio.interceptors.add(_ErrorInterceptor());
  }
}

class _LangInterceptor extends Interceptor {
  @override
  void onRequest(RequestOptions options, RequestInterceptorHandler handler) {
    // Inject lang into every request automatically
    final lang = sl<AppSettingsCubit>().state.locale.languageCode;
    options.queryParameters['lang'] = lang;
    handler.next(options);
  }
}
```
